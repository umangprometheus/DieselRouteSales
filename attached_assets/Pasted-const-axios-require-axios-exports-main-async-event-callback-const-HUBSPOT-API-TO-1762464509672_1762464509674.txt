const axios = require('axios');
exports.main = async (event, callback) => {
  const HUBSPOT_API_TOKEN =
    process.env.HUBSPOT_PRIVATE_APP_TOKEN || 'pat-na1-38796b4c-3a70-4949-95bf-01ebffc0f5da';
  const headers = {
    Authorization: `Bearer ${HUBSPOT_API_TOKEN}`,
    'Content-Type': 'application/json',
  };
  const contactId = event.inputFields['contact_id'];
  if (!contactId) {
    return callback({ outputFields: { success: false, error: 'Contact ID missing' } });
  }
  /* contact + latest meeting */
  const { data: contact } = await axios.get(
    `https://api.hubapi.com/crm/v3/objects/contacts/${contactId}?properties=lastname`,
    { headers }
  );
  const lastName = contact.properties.lastname;
  const { data: mtgResp } = await axios.get(
    `https://api.hubapi.com/engagements/v1/engagements/associated/contact/${contactId}`,
    { headers }
  );
  
  const now = new Date();
  const cutoff = new Date(now.getTime() - 21 * 24 * 60 * 60 * 1000); // 21 days ago
  const meetings = mtgResp.results
    .filter(e => e.engagement.type === 'MEETING')
    .sort((a, b) => new Date(b.engagement.createdAt) - new Date(a.engagement.createdAt))

  if (!meetings.length) {
    return callback({ outputFields: { success: false, error: 'No meetings found' } });
  }
  const mtg = meetings[0];
  const mtgId = mtg.engagement.id;
  /* Scheduled-By email */
  const desc =
    mtg.metadata.description || mtg.metadata.body || '';
  const m = desc.match(/Scheduled By:\s*([^\s]+@[^\s]+)\b/i);
  const schedEmail = m ? m[1].toLowerCase() : null;
  /* creator name logic */
  const ownerId = mtg.engagement.ownerId;
  const createdBy = mtg.engagement.createdBy || ownerId;
  let creatorName = schedEmail ? schedEmail : `User ${createdBy}`;
  if (!schedEmail) {
    try {
      const u = await axios.get(
        `https://api.hubapi.com/settings/v3/users/${createdBy}`,
        { headers }
      );
      creatorName = `${u.data.firstName} ${u.data.lastName}`;
    } catch {
      try {
        const o2 = await axios.get(
          `https://api.hubapi.com/owners/v2/owners/${createdBy}`,
          { headers }
        );
        creatorName = `${o2.data.firstName} ${o2.data.lastName}`;
      } catch {
        try {
          const o3 = await axios.get(
            `https://api.hubapi.com/crm/v3/owners/${createdBy}`,
            { headers }
          );
          creatorName = `${o3.data.firstName} ${o3.data.lastName}`;
        } catch (error) {
          console.log(error)
        }
      }
    }
  }
  /* deal basics */
  const rawTitle = mtg.metadata.title || '';
  const title = rawTitle.includes('Meeting - ')
    ? rawTitle.split('Meeting - ')[1]
    : (rawTitle || 'Meeting');
  const startUTC = new Date(mtg.metadata.startTime);
  const dateStr = new Date(mtg.metadata.startTime).toLocaleDateString(
    'en-US',
    { month: '2-digit', day: '2-digit', year: '2-digit', timeZone: 'America/New_York' }
  );
  const dealName = `${lastName} - ${title} - ${dateStr}`;

  /* Check for existing deal */
  let existingDealId = null;
  let dealId = null;
  try {
    const assocResp = await axios.get(
      `https://api.hubapi.com/crm/v3/objects/contacts/${contactId}/associations/deals`,
      { headers }
    );
    const associatedDeals = assocResp.data.results.map((r) => r.id);
    if (associatedDeals.length) {
      const candidateDeals = [];
      for (const dId of associatedDeals) {
        const { data: deal } = await axios.get(
          `https://api.hubapi.com/crm/v3/objects/deals/${dId}?properties=dealname,pipeline,dealstage,hs_createdate`,
          { headers }
        );
        if (deal.properties.pipeline == 'default' && deal.properties.dealstage == 'qualifiedtobuy') {
          candidateDeals.push({ ...deal, id: dId });
        }
      }
      if (candidateDeals.length) {
        candidateDeals.sort((a, b) => new Date(b.properties.hs_createdate) - new Date(a.properties.hs_createdate));
        existingDealId = candidateDeals[0].id;
      }
    }
  } catch (error) {
    console.log('Error checking existing deals:', error);
  }

  if (existingDealId) {
    const midnightUTC = new Date(Date.UTC(startUTC.getUTCFullYear(), startUTC.getUTCMonth(), startUTC.getUTCDate()));
    const reschedDealName = `${lastName} - ${title} (rescheduled) - ${dateStr}`;
    /* Update existing deal */
    await axios.patch(
      `https://api.hubapi.com/crm/v3/objects/deals/${existingDealId}`,
      {
        properties: {
          dealname: reschedDealName,
          original_meeting_date_time: startUTC.toISOString(),
          hubspot_owner_id: ownerId,
          rescheduled_date: midnightUTC.getTime().toString(),
          meeting_created_by: creatorName,
          rescheduled_by: creatorName,
        }
      },
      { headers }
    );
    dealId = existingDealId;
  } else {
    /* Create new deal */
    const dealRes = await axios.post(
      'https://api.hubapi.com/crm/v3/objects/deals',
      {
        properties: {
          dealname: dealName,
          pipeline: '7311655',
          dealstage: '21286657',
          hubspot_owner_id: ownerId,
          original_meeting_date_time: startUTC.toISOString(),
          meeting_created_by: creatorName
        }
      },
      { headers }
    );
    dealId = dealRes.data.id;
    /* Associate contact to new deal */
    await axios.post(
      'https://api.hubapi.com/crm/v3/associations/contacts/deals/batch/create',
      { inputs: [{ from: { id: contactId }, to: { id: dealId }, type: 'contact_to_deal' }] },
      { headers }
    );
  }

  /* Associate meeting to deal (new or existing) */
  await axios.put(
    `https://api.hubapi.com/engagements/v1/engagements/${mtgId}/associations/deal/${dealId}`,
    {},
    { headers }
  );



  callback({ outputFields: { success: true, dealId, meetingId: mtgId } });
};
